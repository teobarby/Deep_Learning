\section{Introduzione al deep learning}
% ==================================================================

\begin{frame}{Che cos'è il deep learning}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Machine learning classico}
        Le \textbf{feature} sono progettate a mano da un esperto; il modello
        impara solo la funzione che va dalle feature all'output.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Deep learning}
        Le feature vengono \textbf{apprese insieme all'obiettivo}
        (\emph{representational learning}), direttamente dai dati grezzi.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{itemize}
    \item Funzioni complesse ottenute \textbf{componendo} (\emph{nesting}) funzioni
          semplici, in una sequenza di layer sempre più \textbf{astratti}.
    \item Approcci principali: reti \textbf{feedforward}, \textbf{convoluzionali}
          (CNN), \textbf{ricorrenti} (RNN, LSTM), \textbf{Transformer},
          autoencoder, GAN.
  \end{itemize}
\end{frame}

\begin{frame}{Perché proprio adesso}
  Le reti neurali esistono da decenni: il salto è avvenuto quando tre fattori si
  sono combinati (\emph{scale drives deep learning}).
  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Dati}
        Dataset enormi ed etichettati (ImageNet, Object365): le reti grandi
        continuano a migliorare dove i modelli classici si fermano.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Calcolo}
        Le \textbf{GPU} rendono veloci le operazioni matriciali: si possono
        provare più modelli e architetture.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Algoritmi}
        Nuovi metodi, come il passaggio da \textbf{sigmoide a ReLU}, rendono
        addestrabili le reti profonde.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    \small Nel progetto: 14.361 immagini, una GPU consumer e backbone ResNet
    pre-addestrate, cioè tutti e tre i fattori.
  \end{center}
\end{frame}

\begin{frame}{Il neurone}
  \begin{block}{Modello del neurone}
    Il neurone calcola una \textbf{combinazione lineare} degli input e vi applica
    una \textbf{funzione di attivazione} $g$:
    \[ z = \sum_{j=1}^{n} \theta_j x_j + \theta_0 , \qquad a = g(z) \]
  \end{block}
  \begin{itemize}
    \item $\theta_j$ = \textbf{pesi} (quanto conta ogni input), $\theta_0$ =
          \textbf{bias}.
    \item Con il trucco della \emph{bias unit} $x_0=1$ si scrive compattamente
          $z = \theta^{\top}\mathbf{x}$.
    \item Con $g$ = logistica il neurone è esattamente la \textbf{regressione
          logistica}: $h_\theta(\mathbf{x}) = \sigma(\theta^{\top}\mathbf{x})$.
    \item Senza $g$ il neurone è solo un \textbf{prodotto scalare}: è $g$ a dare
          la \alert{non linearità}.
  \end{itemize}
\end{frame}

\begin{frame}{La rete multilayer}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item I neuroni si organizzano in \textbf{layer}: input, uno o più layer
              \textbf{nascosti}, output.
        \item Rete \textbf{deep-$L$}: $L$ = numero di layer nascosti + output.
        \item Ogni unità di un layer riceve tutte le attivazioni del layer
              precedente (layer \emph{fully connected}).
        \item $\Theta^{(l)}$ = matrice dei pesi che collega il layer $l$ al
              layer $l+1$.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Feature apprese}
        La rete \textbf{scopre da sola} le feature nascoste: la semantica la
        diamo noi. Es.\ prezzo di una casa: un'unità nascosta che pesa molto
        CAP e reddito e poco il resto può rappresentare la ``qualità della
        zona''.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Forward propagation}
  L'output di un layer è l'input del successivo:
  \[ a_i^{(l)} = g\Big( \sum_k \theta_{ik}^{(l-1)} a_k^{(l-1)} \Big)
     \qquad\text{cioè}\qquad
     \mathbf{z}^{(l)} = \Theta^{(l-1)} \mathbf{a}^{(l-1)}, \quad
     \mathbf{a}^{(l)} = g\big( \mathbf{z}^{(l)} \big) \]
  {\small $\Theta^{(l-1)}$ collega il layer $l-1$ al layer $l$; il bias è incluso
  tramite la \emph{bias unit} $a_0^{(l-1)}=1$.}
  \vspace{0.2cm}
  \begin{block}{Dall'input all'output}
    \[ \mathbf{a}^{(1)} = \mathbf{x} \;\to\; \mathbf{a}^{(2)} \;\to\; \cdots
       \;\to\; \mathbf{a}^{(L+1)} = h_\Theta(\mathbf{x}) = \hat{y} \]
    Ogni passo è un \textbf{prodotto matrice-vettore} seguito da $g$ applicata
    elemento per elemento: per questo le GPU sono così efficaci.
  \end{block}
  \begin{center}
    \small Una CNN e un Transformer fanno la stessa cosa, cambiando solo
    \emph{quali} pesi collegano quali unità.
  \end{center}
\end{frame}

\begin{frame}{Perché non lineari, perché profonde}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Perché non lineari?}
        Con sole attivazioni \textbf{lineari} la composizione di layer è ancora
        lineare: una rete profonda equivale a \textbf{un solo neurone}. Con
        attivazioni non lineari la rete diventa un
        \alert{approssimatore universale di funzioni}.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Perché profonde?}
        \begin{itemize}
          \item \textbf{Apprendimento gerarchico}: bordi $\to$ parti di volto
                $\to$ volti; forme d'onda $\to$ fonemi $\to$ parole $\to$ frasi.
          \item \textbf{Teoria dei circuiti}: una funzione calcolabile con una
                rete profonda e piccola ($O(\log n)$ layer) richiede a una rete
                poco profonda un numero di unità \textbf{esponenziale}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    \small Nel detector si vede la gerarchia: la ResNet parte dai bordi e arriva
    a feature che riconoscono sedie e persone.
  \end{center}
\end{frame}

\begin{frame}{Funzioni di attivazione}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \[ \sigma(z)=\frac{1}{1+e^{-z}} \qquad \mathrm{ReLU}(z)=\max(0,z) \]
      \begin{itemize}
        \item \textbf{Layer nascosti}: logistica fino agli anni '90, poi
              $\tanh$, oggi \textbf{ReLU}.
        \item \textbf{Output regressione}: lineare / identità.
        \item \textbf{Output binario}: sigmoide (logistica).
        \item \textbf{Output multiclasse}: \textbf{softmax}.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\textwidth]{attivazioni.pdf}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \[ \mathrm{softmax}(\mathbf{z})_k=\frac{e^{z_k}}{\sum_j e^{z_j}}
     \quad\text{(la logistica ne è il caso con due classi)} \]
\end{frame}

\begin{frame}{Pro e contro delle attivazioni}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Sigmoidi (logistica e tanh)}
        \begin{itemize}
          \item[+] Derivabili ovunque; la logistica ha range $[0,1]$, adatto
                alle probabilità; $\tanh$ è centrata in 0.
          \item[--] \textbf{Saturazione}: per $|z|$ grande la derivata è quasi 0
                e i pesi in ingresso smettono di aggiornarsi.
          \item[--] La logistica è sempre positiva: i pesi di un neurone si
                aggiornano tutti nella \textbf{stessa direzione}.
          \item[--] Mai esattamente 0: nessuna \textbf{sparsità}; serve un
                esponenziale.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{ReLU}
        \begin{itemize}
          \item[+] Semplicissima da calcolare, anche la derivata.
          \item[+] \textbf{Non satura} per $z>0$: il gradiente scorre bene.
          \item[+] Produce zeri veri: \textbf{sparsità}, cioè parti della rete
                specializzate su input diversi.
          \item[--] \emph{Dying ReLU}: un neurone sempre a 0 non impara più
                $\to$ varianti \textbf{Leaky ReLU} ed \textbf{ELU}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \begin{center}
    ReLU + inizializzazione di He è lo stato dell'arte per reti feedforward e
    convoluzionali: la usano sia la ResNet sia il nostro collo del detector.
  \end{center}
\end{frame}

\begin{frame}{La funzione di costo}
  La rete produce $\hat{y}$; la \textbf{loss} misura quanto sbaglia su un
  esempio, il \textbf{costo} $J$ è la media su tutti gli $m$ esempi:
  \[ J(\Theta)=\frac{1}{m}\sum_{i=1}^{m} L\!\left(\hat{y}^{(i)}, y^{(i)}\right) \]
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{Regressione}
        Errore quadratico
        \[ L = \tfrac{1}{2}\left(\hat{y}-y\right)^2 \]
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Binaria}
        Cross-entropy binaria
        \[ L = -y\log\hat{y} - (1-y)\log(1-\hat{y}) \]
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Multiclasse}
        Cross-entropy (softmax)
        \[ L = -\sum_k y_k \log \hat{y}_k \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  {\footnotesize Il detector le usa \textbf{tutte e tre}: quadratica per le
  coordinate delle box, binaria per ``c'è un oggetto'', multiclasse per la
  classe. Il classificatore di sentiment usa la cross-entropy su due classi.}
\end{frame}

\section{Addestrare una rete profonda}
% ==================================================================

\begin{frame}{Gradient descent}
  \begin{block}{Regola di aggiornamento}
    Si aggiornano i parametri nella direzione \textbf{opposta al gradiente}:
    \[ \theta := \theta - \alpha \frac{\partial J}{\partial \theta} \]
    con $\alpha$ = \textbf{learning rate} (ampiezza del passo).
  \end{block}
  \begin{itemize}
    \item $\alpha$ troppo grande $\to$ si oscilla o si diverge; troppo piccolo
          $\to$ convergenza lentissima.
    \item $J$ di una rete profonda \textbf{non è convessa}: si arriva a un minimo
          locale, che in pratica è quasi sempre abbastanza buono.
    \item Resta da calcolare $\partial J / \partial \theta$ per \textbf{milioni}
          di pesi in modo efficiente: è il compito della backpropagation.
  \end{itemize}
\end{frame}

\begin{frame}{Backpropagation}
  Si propaga l'errore \textbf{all'indietro}, dal layer di output verso l'input,
  applicando la \emph{regola della catena}. Per ogni layer si calcola l'errore
  $\boldsymbol{\delta}^{(l)}$ delle sue unità:
  \begin{align*}
    \boldsymbol{\delta}^{(L+1)} &= \mathbf{a}^{(L+1)} - \mathbf{y}
      && \text{errore sull'output} \\
    \boldsymbol{\delta}^{(l)} &= \big(\Theta^{(l)}\big)^{\!\top}
      \boldsymbol{\delta}^{(l+1)} \odot g'\big(\mathbf{z}^{(l)}\big)
      && \text{errore riportato indietro} \\
    \frac{\partial J}{\partial \theta_{ij}^{(l)}} &= a_j^{(l)}\,\delta_i^{(l+1)}
      && \text{gradiente di ogni peso (mediato sugli esempi)}
  \end{align*}
  \begin{itemize}
    \item L'errore di un'unità nascosta usa la \textbf{derivata della sua
          funzione di attivazione} $g'$.
    \item Costo: circa quello di una forward propagation. Un passo di training
          = forward + backward + aggiornamento.
    \item In PyTorch il calcolo è automatico (\emph{autograd}): noi scriviamo
          solo la forward e la loss.
  \end{itemize}
\end{frame}

\begin{frame}{Gradienti instabili: vanishing ed exploding}
  Il gradiente di un layer poco profondo è un \textbf{prodotto} di tanti
  fattori, uno per ogni layer successivo: pesi $\times$ derivate
  dell'attivazione.
  \[ \frac{\partial J}{\partial \mathbf{a}^{(1)}} \;\propto\;
     \prod_{l} \theta^{(l)}\, g'\big(z^{(l)}\big) \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Vanishing}
        Se ogni fattore è $<1$ il prodotto tende a 0. Tipico con la logistica:
        $\sigma'(z) \le \tfrac{1}{4}$. I primi layer smettono di imparare.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Exploding}
        Se i pesi sono grandi, ogni fattore è $>1$ e il prodotto cresce
        esponenzialmente verso i primi layer. Il training diverge.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  In entrambi i casi layer diversi imparano a \textbf{velocità diverse}.
  Rimedi: \textbf{ReLU}, buona \textbf{inizializzazione}, batch normalization,
  connessioni residue (ResNet).
\end{frame}

\begin{frame}{Inizializzazione dei pesi}
  Con input standardizzati e pesi da $\mathcal{N}(0,1)$, la somma
  $z=\sum_{i=1}^{n}\theta_i x_i$ di $n$ termini ha
  \[ \mathbb{E}[z]=0, \qquad \mathrm{Var}(z) = n \]
  Per $n$ grande $z$ \textbf{esplode} (o, all'indietro, svanisce), e con le
  sigmoidi finisce subito in saturazione.
  \begin{block}{Soluzione: riscalare i pesi di ogni layer}
    Si moltiplicano i pesi estratti per $\sqrt{c/n^{(l-1)}}$, con $n^{(l-1)}$ =
    numero di input del layer, così che $\mathrm{Var}(z)\approx 1$ in ogni layer:
    \begin{itemize}
      \item $c=1$: attivazione lineare, o \textbf{Xavier} per $\tanh$;
      \item $c=2$: \textbf{He}, per la \textbf{ReLU} (che azzera metà degli
            input, dimezzando la varianza).
    \end{itemize}
  \end{block}
  \begin{center}
    \small Nel progetto i layer nuovi (collo e testa del detector) partono
    dall'inizializzazione standard di PyTorch; il backbone parte dai pesi
    pre-addestrati.
  \end{center}
\end{frame}

\begin{frame}{Overfitting e regolarizzazione L2}
  Una rete grande può \textbf{memorizzare} il training set: errore basso sul
  training, alto su dati nuovi (\emph{overfitting}, alta varianza).
  \begin{block}{Regolarizzazione L2}
    Si penalizzano i pesi grandi aggiungendo un termine al costo:
    \[ J_{\text{reg}}(\Theta) = J(\Theta) + \frac{\lambda}{2m}
       \sum_{l}\sum_{i,j} \big(\theta_{ij}^{(l)}\big)^2 \]
    Nel gradient descent diventa $\theta := \theta\big(1-\alpha\tfrac{\lambda}{m}\big)
    - \alpha\tfrac{\partial J}{\partial\theta}$: a ogni passo i pesi si
    ``restringono'' (\textbf{weight decay}).
  \end{block}
  \begin{itemize}
    \item \textbf{Perché funziona}: pesi piccoli $\Rightarrow$ $z$ vicino a 0
          $\Rightarrow$ l'attivazione lavora nella zona quasi \textbf{lineare}
          $\Rightarrow$ la rete può rappresentare solo funzioni più semplici.
    \item $\lambda$ grande $\to$ reti più semplici e sparse, meno overfitting;
          troppo grande $\to$ underfitting.
  \end{itemize}
\end{frame}

\begin{frame}{Dropout, data augmentation, early stopping}
  \small
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \begin{block}{Dropout}
        Durante il training ogni nodo viene \textbf{eliminato} con probabilità
        $p$, diversa a ogni esempio.
        \begin{itemize}
          \item Si addestrano reti più piccole.
          \item Nessuna unità può contare su una singola feature: i pesi si
                distribuiscono.
          \item \textbf{Mai a test time.}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Data augmentation}
        Si creano nuovi esempi trasformando quelli esistenti: flip, crop,
        zoom, rotazioni, variazioni di colore.
        \\[4pt]
        Naturale in visione: un'auto riflessa è ancora un'auto.
      \end{block}
    \end{column}
    \begin{column}{0.34\textwidth}
      \begin{block}{Early stopping}
        Si ferma il training quando l'errore sul \textbf{validation}
        ricomincia a salire.
        \\[4pt]
        Limite: ottimizza il costo e riduce la varianza \emph{insieme}, mentre
        L2 separa i due obiettivi (\emph{ortogonalità}).
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    Nel progetto: \textbf{augmentation} nel detector (flip, colore, zoom e
    traslazione), \textbf{dropout} dentro DistilBERT, e in entrambi il
    checkpoint scelto sul validation, che è una forma di early stopping.
  \end{center}
\end{frame}

\begin{frame}{Mini-batch gradient descent}
  Con big data calcolare il gradiente su \textbf{tutti} gli $m$ esempi prima di
  ogni passo è proibitivo. Si divide il training set in mini-batch e si
  aggiornano i pesi dopo \textbf{ogni} mini-batch; un'\textbf{epoca} è un
  passaggio completo sul training set.
  \vspace{0.2cm}
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    Dimensione del batch & Nome & Comportamento \\
    \midrule
    $m$ & batch GD & passi precisi ma lenti \\
    $1$ & stochastic GD (SGD) & veloce ma rumoroso, perde la vettorizzazione \\
    $64$--$512$ & \textbf{mini-batch GD} & compromesso: vettorizzato e frequente \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.2cm}
  \begin{itemize}
    \item Batch piccoli: convergenza rapida ma soluzione meno precisa; batch
          grandi: più lenti ma più stabili.
    \item Il mini-batch deve \textbf{stare in memoria} della GPU.
          \emph{Nel progetto: 16 immagini per il detector, 16 frasi per BERT.}
  \end{itemize}
\end{frame}

\begin{frame}{Medie mobili esponenziali}
  Sono il mattone degli ottimizzatori moderni:
  \[ v_t = \beta\, v_{t-1} + (1-\beta)\, x_t \]
  \begin{itemize}
    \item È una media sugli ultimi $\approx \frac{1}{1-\beta}$ valori: con
          $\beta=0.9$ circa 10, con $\beta=0.98$ circa 50.
    \item I termini più vecchi pesano meno, perché il loro peso $\beta^{k}$
          decresce con la distanza.
    \item $\beta$ grande $\to$ curva più liscia ma in ritardo; $\beta$ piccolo
          $\to$ segue il rumore.
  \end{itemize}
  \begin{block}{Bias correction}
    Partendo da $v_0=0$ i primi valori sono troppo bassi. Si corregge con
    \[ \hat{v}_t = \frac{v_t}{1-\beta^{t}} \]
    All'inizio il fattore normalizza la somma dei pesi; per $t$ grande vale
    circa 1 e non ha effetto.
  \end{block}
\end{frame}

\begin{frame}{Momentum, RMSprop, Adam}
  \small
  Si scrive $d\theta$ per il gradiente $\partial J/\partial\theta$ calcolato sul
  mini-batch.
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{Momentum}
        Media mobile del \textbf{gradiente}:
        \[ v = \beta_1 v + (1-\beta_1)\, d\theta \]
        \[ \theta := \theta - \alpha\, v \]
        Smorza le oscillazioni e aiuta a \textbf{uscire dai minimi locali}.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{RMSprop}
        Media mobile del \textbf{quadrato}:
        \[ s = \beta_2 s + (1-\beta_2)\, d\theta^2 \]
        \[ \theta := \theta - \alpha\,\frac{d\theta}{\sqrt{s}+\varepsilon} \]
        Passi più piccoli dove il gradiente oscilla molto, più grandi dove è
        piccolo.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Adam}
        \emph{ADAptive Momentum}: le due insieme, con bias correction:
        \[ \theta := \theta - \alpha\,\frac{\hat{v}}{\sqrt{\hat{s}}+\varepsilon} \]
        Valori tipici: $\beta_1=0.9$, $\beta_2=0.999$,
        $\varepsilon=10^{-8}$; si regola solo $\alpha$.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    Adam è l'ottimizzatore usato in \textbf{entrambi} gli esercizi (nell'Es.\ 2
    nella variante \textbf{AdamW}), con $\alpha=10^{-4}$ nel detector e
    $2\cdot10^{-5}$ per BERT.
  \end{center}
\end{frame}

\begin{frame}{Batch Normalization}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Idea}
        Come si normalizzano gli input, si normalizzano anche i valori dei
        \textbf{nodi interni} (prima dell'attivazione), con media e varianza
        del \textbf{mini-batch}:
        \[ \mu = \frac{1}{m}\sum_i z_i, \qquad
           \sigma^2 = \frac{1}{m}\sum_i (z_i-\mu)^2 \]
        \[ z_{\text{norm}} = \frac{z-\mu}{\sqrt{\sigma^2+\varepsilon}}, \qquad
           \tilde{z} = \gamma\, z_{\text{norm}} + \beta \]
        $\gamma,\beta$ si apprendono: media 0 e varianza 1 non sono sempre
        ottimali (la sigmoide lì è quasi lineare).
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Contrasta il \textbf{covariate shift}: l'input di un layer profondo
              dipende da tutti i pesi precedenti e cambia di continuo durante il
              training. BN ne fissa media e varianza.
        \item Il rumore delle statistiche del mini-batch fa anche da
              \textbf{regolarizzazione} (più forte con batch piccoli).
        \item La normalizzazione assorbe il bias: il layer prima della BN non
              ne ha bisogno (nel detector: 	exttt{Conv2d(..., bias=False)}).
        \item \textbf{A test time} si usano le medie mobili esponenziali di
              $\mu$ e $\sigma^2$ accumulate durante il training.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \begin{center}
    Usata in ogni blocco convoluzionale del detector:
    \texttt{Conv $\to$ BN $\to$ ReLU}. Il Transformer usa la variante
    \emph{layer norm}.
  \end{center}
\end{frame}

\begin{frame}{Dalla teoria ai due esercizi}
  \begin{block}{Quello che serve a entrambi}
    Rete profonda con attivazioni \textbf{ReLU}, loss di \textbf{cross-entropy},
    backpropagation, mini-batch con \textbf{Adam}, normalizzazione,
    regolarizzazione e selezione del modello sul \textbf{validation}.
  \end{block}
  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Object detection}
        Immagini $\to$ rete \textbf{convoluzionale} (CNN), che sfrutta la
        struttura spaziale dei pixel. \emph{Architettura YOLO.}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Sentiment analysis}
        Frasi $\to$ \textbf{Transformer}, che mette in relazione ogni parola con
        tutte le altre. \emph{Modello BERT.}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    \small In entrambi i casi si parte da una rete \textbf{pre-addestrata}
    (\emph{transfer learning}).
  \end{center}
\end{frame}

\section{Object Detection (YOLO)}

\begin{frame}{Reti convoluzionali: la convoluzione}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Una rete densa su immagini grandi è \textbf{ingestibile} (troppi pesi).
      \vspace{0.2cm}

      La \textbf{convoluzione} fa scorrere un piccolo \textbf{filtro} (kernel)
      sull'immagine, calcolando un prodotto scalare a ogni posizione $\to$
      produce una \emph{feature map}.
      \vspace{0.2cm}
      \begin{block}{Filtri appresi, non progettati}
        Un filtro può rilevare i \textbf{bordi} (verticali, orizzontali).
        Invece di costruirlo a mano (Sobel, Scharr), si \alert{apprendono i pesi
        del filtro} con la backpropagation.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\textwidth]{cnn_edge.pdf}\\[4pt]
      {\footnotesize Un filtro verticale evidenzia i bordi verticali.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{CNN: padding, stride, pooling}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Iperparametri della convoluzione}
        \begin{itemize}
          \item \textbf{Padding}: bordo di zeri; \emph{same} mantiene la
                dimensione, \emph{valid} la riduce.
          \item \textbf{Stride}: di quanti passi scorre il filtro.
          \item \textbf{Più filtri} $\to$ più feature map (profondità).
                Su immagini RGB: convoluzione su ogni canale, poi somma.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Pooling}
        Il \textbf{max pooling} tiene il valore massimo di ogni regione:
        riduce la dimensione e agisce da rilevatore di feature.
      \end{block}
      \vspace{0.1cm}
      \textbf{Architettura tipica}:
      \[ \text{CONV} \to \text{POOL} \to \dots \to \text{FC} \]
      Andando in profondità: più filtri, dimensioni spaziali più piccole.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Perché la convoluzione funziona}
  \begin{block}{Due proprietà chiave}
    \begin{itemize}
      \item \textbf{Parameter sharing}: un rilevatore di feature (es.\ un bordo)
            è utile in \textbf{tutta} l'immagine $\to$ pochissimi parametri, che
            \emph{non dipendono} dalla dimensione dell'immagine.
      \item \textbf{Sparsità delle connessioni}: ogni output dipende solo da una
            \textbf{piccola regione} dell'input $\to$ meno parametri, meno
            overfitting.
    \end{itemize}
  \end{block}
  \begin{center}
    \small Nel detector: la ResNet riduce $416\times416$ fino alla griglia
    $26\times26$ (stride 16); il ``collo'' aggiunge due blocchi
    \texttt{Conv $\to$ BN $\to$ ReLU} a risoluzione costante.
  \end{center}
\end{frame}

\begin{frame}{Il problema: localizzazione con bounding box}
  \begin{block}{Object detection}
    Data un'immagine, individuare \textbf{quali} oggetti contiene e \textbf{dove}
    si trovano, disegnando una \emph{bounding box} attorno a ciascuno.
  \end{block}
  \vspace{0.3cm}
  \textbf{Dataset: Object365} (365 classi, 600k immagini, 10M box).
  \begin{itemize}
    \item Volume incompatibile con un PC $\Rightarrow$ \alert{subsampling} su due assi.
    \item \textbf{Immagini}: 15 shard scaricati $\to$ \textbf{14.361} immagini (119.215 box),
          divise in train 11.467 / validation 1.395 / test 1.499.
    \item \textbf{Classi}: da 365 a \textbf{8}, scelte \emph{misurando} frequenza e dimensione:
          Person, Chair, Table, Cabinet, Car, Lamp, Picture, Monitor.
    \item Categorie ``sorelle'' \textbf{unite} (Car + SUV + Van + Pickup): altrimenti un SUV
          sarebbe sfondo accanto a un'auto identica etichettata come oggetto.
  \end{itemize}
  \begin{center}
    \small Tecnica scelta: \textbf{YOLO}. Griglia, anchor, loss, IoU e NMS sono
    \emph{implementati a mano} (nessuna libreria di detection).
  \end{center}
\end{frame}

\begin{frame}{Dalla sliding window a YOLO}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Sliding window}: si fa scorrere una finestra e si classifica ogni ritaglio.
      \begin{itemize}
        \item Problema principale: \alert{costo computazionale}.
        \item Soluzione: implementarla \textbf{convoluzionalmente}.
      \end{itemize}
      \vspace{0.2cm}
      \textbf{YOLO}: griglia $S\times S$; ogni oggetto assegnato a \textbf{una cella}
      (quella del suo centro), che ne predice box e classe.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Anchor box}
        Per ogni cella, $n$ forme predefinite. Se più oggetti cadono nella stessa
        cella, ognuno va all'anchor più simile per forma (IoU).
      \end{block}
      \centering
      \includegraphics[width=0.86\textwidth]{anchor.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Distribuzione del dataset}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \vspace{0.4cm}
      Sottoinsieme \textbf{sbilanciato}: \textbf{Person} è il 51\% delle box,
      \textbf{Monitor} meno del 2\% $\Rightarrow$ prestazioni attese
      \alert{inferiori sulle classi rare}.
      \vspace{0.3cm}

      \begin{block}{Regioni ignorate (\emph{don't care})}
        Box \textbf{crowd} (una box su un gruppo di oggetti, il 15\% delle
        annotazioni) e oggetti \textbf{più piccoli di una cella}: etichetta
        ``?'', niente obiettivi né penalità. Cancellarli insegnerebbe che lì \emph{non c'è nulla}.
      \end{block}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\textwidth]{classi_new.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Architettura della rete}
  Backbone convoluzionale con downsampling $\times 32$, seguito da una
  \textbf{testa di detection} $1\times1$ scritta da noi.
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      {\footnotesize
      \begin{semiverbatim}
Input  3 x 416 x 416
  ResNet34 troncata a layer3 (stride 16)
             -> 256 x 26 x 26   (griglia S)
  Collo: 2 x (conv 3x3 + BN + ReLU)
  Testa 1x1 -> A*(5+C) = 65 canali
Output (5, 26, 26, 13)
      \end{semiverbatim}}
      \vspace{0.1cm}
      \begin{itemize}
        \item Rete \textbf{completamente convoluzionale}.
        \item \textbf{3.380} box candidate per immagine ($5\times26\times26$).
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Backbone: transfer learning}
        \alert{\textbf{ResNet34 pre-addestrata}} su ImageNet e fine-tunata.
        Alternative provate: ResNet18 (--0,05 di mAP) e un backbone scritto
        interamente da zero.
      \end{block}
      {\footnotesize Anchor, target, loss, IoU e NMS restano \textbf{implementati a
      mano}: cambia solo l'estrattore di feature.}
      \vspace{0.15cm}
      {\footnotesize Il \textbf{troncamento} decide il compromesso: stride piccolo
      $\to$ griglia fine ma campo recettivo ridotto; stride grande $\to$ vede più
      contesto ma celle grossolane.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{La loss YOLO}
  La rete predice \emph{logit grezzi}, decodificati come (YOLOv2):
  \[
    b_x=\sigma(t_x)+c_x,\quad b_y=\sigma(t_y)+c_y,\quad
    b_w=a_w e^{t_w},\quad b_h=a_h e^{t_h}
  \]
  \begin{block}{Quattro componenti}
    \centering
    \begin{tabular}{llll}
      \toprule
      Componente & Funzione & Peso & Si applica a \\
      \midrule
      Coordinate & MSE & $\lambda=5$ & anchor responsabili \\
      Objectness & BCE & $1$ & anchor responsabili \\
      No-object & BCE & $\lambda=10$ & anchor vuoti \emph{e non ignorati} \\
      Classe & Cross-entropy & $1$ & anchor responsabili \\
      \bottomrule
    \end{tabular}
  \end{block}
  Su 3.380 box quasi tutte sono vuote: il termine no-object è normalizzato come
  \textbf{media sulle celle vuote} (non dipende dalla griglia) e pesato $\lambda=10$,
  il rapporto del YOLO originale. Troppo basso $\to$ oggetti ovunque; troppo alto
  $\to$ \alert{``non c'è mai niente''}.
\end{frame}

\begin{frame}{Inferenza: non-max suppression}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Ogni box: \textbf{score} $=$ objectness $\times$ prob.\ classe.
      \begin{enumerate}
        \item Scarta le box sotto soglia (0,25).
        \item \textbf{NMS} per classe.
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Non-max suppression}
        {\footnotesize
        ordina le box per score\\
        finché restano box:\\
        \quad prendi quella con score max $\to$ predizione\\
        \quad scarta le altre con $\text{IoU}>$ soglia}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  NMS \emph{per classe}: due oggetti di classi diverse possono sovrapporsi (una
  persona con un cappello); due box sovrapposte della stessa classe no.
\end{frame}

\begin{frame}{Come si misura un detector}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Una predizione è \textbf{corretta} se ha IoU $\geq 0{,}5$ con un oggetto della
      stessa classe non ancora abbinato; le altre sono falsi positivi.
      \[ P=\frac{TP}{TP+FP}\qquad R=\frac{TP}{N_{ogg}}\qquad F_1=\frac{2PR}{P+R} \]
      \begin{alertblock}{La trappola della soglia fissa}
        Lo stesso modello: \textbf{F1 0,145} a soglia 0,25 e \textbf{0,249} a 0,65.
        Confrontare modelli a soglia fissa è ingannevole.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{mAP@0.5}
        Si fa variare la soglia, si costruisce la curva \textbf{precision--recall} e
        se ne prende l'\textbf{area} (AP), mediata sulle classi.
        \textbf{Non dipende dalla soglia} $\Rightarrow$ è la metrica con cui si
        confrontano le varianti e si sceglie il checkpoint.
      \end{block}
      {\footnotesize La soglia d'uso (0,9) è poi quella che massimizza l'F1 sul
      validation, salvata nel checkpoint.}
    \end{column}
  \end{columns}
\end{frame}

\section{Sentiment Analysis (Transformer)}
% ==================================================================

\begin{frame}{Il problema: classificare frasi}
  \begin{block}{Sentiment analysis}
    Classificare una sequenza testuale come \textbf{positiva} o \textbf{negativa},
    usando un'architettura \textbf{transformer}.
  \end{block}
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Dataset}: SST-2 (frasi brevi etichettate pos/neg, da HuggingFace).
    \item Corrisponde alla traccia (``frasi positive o negative''); nessun token Kaggle richiesto.
    \item Tre split: \textbf{train}; \textbf{validation} (2000 frasi ricavate dal
          train) per scegliere il checkpoint; \textbf{test} (le 872 frasi del
          validation ufficiale) usato \emph{una sola volta} per i risultati.
  \end{itemize}
  \begin{center}
    \small Il codice supporta comunque un CSV Kaggle via \texttt{-{}-dataset csv}.
  \end{center}
\end{frame}

\begin{frame}{Dai limiti delle RNN all'attention}
  \textbf{Le RNN} codificano bene la storia recente, ma faticano su contesti
  lontani e sono \textbf{sequenziali} (lente).
  \begin{block}{Esempio delle slide}
    \emph{``I grew up in France, \dots\ thus I speak ??''} --- la parola da predire
    (\emph{French}) dipende da un termine molto \textbf{lontano} nella sequenza.
  \end{block}
  \vspace{0.2cm}
  La risposta è il \textbf{meccanismo di attention}: per produrre la
  rappresentazione di una parola, la si mette in relazione con \textbf{tutte} le
  altre e si pesa quali sono rilevanti, indipendentemente dalla distanza.
\end{frame}

\begin{frame}{Self-attention}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      L'output per ogni parola è una \textbf{somma pesata} delle rappresentazioni
      di tutte le parole; i pesi misurano quanto ognuna è \emph{rilevante}.
      \begin{block}{Proprietà (dalle slide)}
        \begin{itemize}
          \item Calcolo \textbf{parallelo} di tutti gli output (non sequenziale).
          \item \emph{Permutation equivariant}: è più un modello di
                \textbf{insiemi} che di sequenze $\Rightarrow$ serve il
                \textbf{positional encoding} per reintrodurre l'ordine.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=0.92\textwidth]{attention.pdf}\\[3pt]
      {\footnotesize \emph{cattivo} mette attenzione su \emph{non}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Query, Key, Value e multi-head}
  \begin{block}{Tre ruoli, tre proiezioni lineari apprese}
    \begin{itemize}
      \item \textbf{Query}: il vettore che viene confrontato con tutti gli altri.
      \item \textbf{Key}: il vettore contro cui la query è confrontata.
      \item \textbf{Value}: ciò che si somma (pesato) per formare l'output.
    \end{itemize}
  \end{block}
  \[ \mathrm{Attention}(Q,K,V)=\mathrm{softmax}\!\left(\frac{QK^{\top}}{\sqrt{d}}\right)V \]
  \begin{itemize}
    \item Divisione per $\sqrt{d}$: \textbf{scaled} attention (stabilità).
    \item \textbf{Multi-head}: più ``teste'' in parallelo, ciascuna cattura un
          tipo diverso di relazione tra le parole.
  \end{itemize}
\end{frame}

\begin{frame}{Il blocco Transformer}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{Un blocco impila due sotto-strati}
        \begin{enumerate}
          \item \textbf{Multi-head self-attention}
          \item \textbf{Rete feed-forward} (applicata a ogni posizione)
        \end{enumerate}
        Ciascuno con una \textbf{connessione residua} (skip) e una
        \textbf{layer normalization}:
        \[ \mathrm{LayerNorm}\big(x + \mathrm{Sublayer}(x)\big) \]
      \end{block}
      Impilando più blocchi si ottiene il Transformer.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Positional encoding}
        La self-attention è \emph{permutation equivariant}: si \textbf{somma} a
        ogni token un vettore che ne codifica la \textbf{posizione}, così il
        modello conosce l'ordine.
      \end{block}
      \begin{block}{Layer normalization}
        Come la batch norm, ma normalizza sulle \textbf{feature di ogni token}
        (non sul batch).
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Il Transformer e BERT}
  \begin{block}{Transformer}
    Un modello che usa principalmente la \textbf{self-attention} per propagare
    l'informazione tra le unità (token di testo, ma anche pixel: Vision
    Transformer).
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{BERT}: transformer \textbf{encoder-only} \emph{bidirezionale},
      pre-addestrato con il \textbf{masking} (predire token nascosti).
      Adatto a \emph{comprendere} una frase, non a generarla.
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Il nostro uso}
        \textbf{Fine-tuning} (transfer learning): si parte da BERT
        pre-addestrato e si aggiunge una testa di classificazione binaria.
        Modello: \textbf{DistilBERT}, un BERT più leggero (6 strati invece di 12).
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Analisi qualitativa: il caso della negazione}
  Frasi costruite ad hoc:
  \begin{block}{Output del modello}
    {\footnotesize\ttfamily
    {}[positive~~99.9\%] I absolutely loved this film\\
    {}[negative~~99.5\%] the worst purchase I have ever made\\
    {}[positive~~99.8\%] It is not bad at all, quite enjoyable\\
    {}[negative~~75.3\%] predictable but the acting saved it}
  \end{block}
  \begin{itemize}
    \item \emph{``not bad''}: contiene \emph{bad} ma è classificata \textbf{positiva} al 99,8\%.
          L'attention collega \emph{bad} a \emph{not} --- è l'\alert{esempio della slide 11}.
    \item Frase ambigua $\Rightarrow$ confidenza bassa (75\%): il modello \emph{segnala} l'incertezza.
  \end{itemize}
\end{frame}
