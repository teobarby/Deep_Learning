\section{Introduzione al deep learning}
% ==================================================================

\begin{frame}{Che cos'è il deep learning}
  Studio e applicazione di metodi basati su \textbf{reti neurali} per problemi
  ad alto impatto: visione, linguaggio, parlato.
  \vspace{0.1cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Machine learning classico}
        \centering
        {\footnotesize dati $\to$ \textbf{feature definite a mano}
        (selezione, costruzione, trasformazione) $\to$ Learning $\to$ f(X)}
        \\[4pt]
        {\footnotesize La definizione delle feature è manuale: è il
        passo \emph{fondamentale} della pipeline.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Deep learning}
        \centering
        {\footnotesize dati $\to$ trasformazioni \textbf{apprese}
        $z_1 \to z_2 \to \dots \to z_h$ $\to$ Classifier $\to$ f(X)}
        \\[4pt]
        {\footnotesize Rappresentazioni intermedie via via più astratte, es.\
        \emph{pixel $\to$ edge $\to$ motif $\to$ parti $\to$ oggetti}
        (\emph{representational learning}).}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \begin{itemize}
    \item Funzioni complesse calcolate \textbf{annidando} (\emph{nesting})
          funzioni semplici: una sequenza di layer sempre più \textbf{astratti}.
    \item Approcci principali: reti \textbf{feedforward} profonde, reti
          \textbf{convoluzionali} (CNN), reti \textbf{ricorrenti},
          \textbf{Transformer}, ecc.
  \end{itemize}
\end{frame}

\begin{frame}{Il neurone}
  \begin{block}{Modello del neurone}
    Combinazione lineare degli input seguita da una \textbf{funzione di
    attivazione} $\varphi$:
    \[ y = \varphi\Big( \sum_{i} \theta_i x_i \Big) = h_\theta(\mathbf{x}) \]
  \end{block}
  \begin{itemize}
    \item $\theta_1,\dots,\theta_n$ = \textbf{pesi}; $\theta_0$ = \textbf{bias},
          con la \emph{bias unit} $x_0=1$.
    \item Con $\varphi$ = logistica il neurone è la \textbf{regressione
          logistica}.
    \item Senza $\varphi$ il neurone è solo un \textbf{prodotto scalare}: è
          $\varphi$ a dare la \alert{non linearità}.
  \end{itemize}
\end{frame}
\begin{frame}{Nomenclatura}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1em}
        \item I neuroni si organizzano in \textbf{layer}: input, layer \textbf{nascosti}, output.
        \item Rete \textbf{deep-$L$}: $L$ = numero di layer nascosti + output.
        \item $\theta_{ik}^{(l)}$ = peso dall'unità $k$ del layer $l$ all'unità $i$ del layer $l+1$; $\Theta^{(l)}$ ne è la matrice.
      \end{itemize}
    \end{column}

    \begin{column}{0.60\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=0.9\linewidth]{figure/FF_NN.png}
        \label{fig:placeholder}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}




\begin{frame}{Forward propagation}
  L'output di un layer è l'input del successivo:
  \[ a_i^{(1)} = x_i, \qquad
     a_i^{(l)} = \varphi\Big( \sum_k \theta_{ik}^{(l-1)} a_k^{(l-1)} \Big)
     \quad 1 < l \le L \]
  \vspace{-0.2cm}
  \begin{itemize}
    \item In forma vettoriale: $\mathbf{a}^{(l)} = \varphi\big(\Theta^{(l-1)}
          \mathbf{a}^{(l-1)}\big)$, con la bias unit $a_0^{(l-1)}=1$.
    \item L'output della rete è $h_\Theta(\mathbf{x}) = \mathbf{a}^{(L)}$.
    \item Ogni passo è un \textbf{prodotto matrice-vettore} seguito da
          $\varphi$ applicata elemento per elemento.
  \end{itemize}
   \begin{block}{Perché funzioni non lineari?}
    Con attivazioni \textbf{lineari}, la composizione di più layer equivale a un \textbf{singolo layer lineare}. Un modello puramente lineare può solo tracciare iperpiani rigidi nello spazio dei dati, risultando inadeguato.
  \end{block}
\end{frame}


\begin{frame}{Funzioni di attivazione}
  \small
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{tabular}{lll}
        \toprule
        & $\varphi(z)$ & $\varphi'(z)$ \\
        \midrule
        Logistica & $\dfrac{1}{1+e^{-z}}$ & $\varphi(z)\big(1-\varphi(z)\big)$ \\[8pt]
        $\tanh$ & $\dfrac{e^{z}-e^{-z}}{e^{z}+e^{-z}}$ & $1-\varphi(z)^2$ \\[8pt]
        ReLU & $\max(0,z)$ & $0$ se $z<0$, $1$ se $z\ge 0$ \\[4pt]
        Leaky ReLU & $\max(\alpha z,z)$ & $\alpha$ se $z<0$, $1$ se $z\ge 0$ \\[4pt]
        ELU & $\alpha(e^{z}-1)$ se $z<0$, $z$ altrimenti &
              $\varphi(z)+\alpha$ se $z<0$, $1$ altrimenti \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \includegraphics[width=\textwidth]{attivazioni.pdf}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Layer nascosti}: logistica fino agli anni '90, poi $\tanh$,
          oggi \textbf{ReLU}.
    \item \textbf{Output}: lineare per la regressione, logistica per la
          classificazione binaria, \textbf{softmax} per la multiclasse.
  \end{itemize}
\end{frame}
\begin{frame}{Pro e contro delle funzioni di attivazione}
  \small
  \begin{columns}[T, totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{\centering Logistica (Sigmoide) e $\tanh$}
        \begin{itemize}
          \setlength{\itemsep}{0.3em}
          \item[\textcolor{green!60!black}{\textbf{+}}] \textbf{Derivabili} ovunque.
          \item[\textcolor{green!60!black}{\textbf{+}}] Logistica in $[0,1]$ (ottima per probabilità); $\tanh$ è centrata in 0.
          \item[\textcolor{red}{\textbf{--}}] \textbf{Saturazione:} per $|z|$ grande il gradiente tende a 0 (vanishing gradient).
          \item[\textcolor{red}{\textbf{--}}] Logistica sempre positiva (gli aggiornamenti dei pesi oscillano).
          \item[\textcolor{red}{\textbf{--}}] \textbf{Nessuna sparsità} (mai esattamente 0) e costo computazionale elevato ($\exp$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\centering ReLU}
        \begin{itemize}
          \setlength{\itemsep}{0.3em}
          \item[\textcolor{green!60!black}{\textbf{+}}] \textbf{Computazione efficiente} (soluzione veloce e derivata immediata).
          \item[\textcolor{green!60!black}{\textbf{+}}] \textbf{Non satura} per $z > 0$ (evita il vanishing gradient).
          \item[\textcolor{green!60!black}{\textbf{+}}] Genera \textbf{sparsità}: produce zeri effettivi disattivando neuroni non necessari.
          \item[\textcolor{red}{\textbf{--}}] \textbf{Dying ReLU:} se un neurone muore (sempre $0$), non impara più.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Loss e Funzione di Costo}
  \small
  \textbf{Loss} $\ell(\hat{y},y)$: errore sul singolo esempio. \\
  \textbf{Costo} $J(\Theta)$: media dell'errore su $m$ esempi:
  $J(\Theta) = \frac{1}{m}\sum_{i=1}^{m} \ell\big(h_\Theta(\mathbf{x}^{(i)}), y^{(i)}\big)$

  \vspace{0.2cm}

  \begin{center}
    \footnotesize
    \begin{tabular}{lll}
      \toprule
      \textbf{Loss} & \textbf{Ambito} & \textbf{Formula} \\
      \midrule
      \textbf{MSE} & Regressione ($y \in \mathbb{R}$) & $\frac{1}{2}(\hat{y}-y)^2$ \\[2pt]
      \textbf{Cross-Entropy} & Class. Binaria ($y \in \{0,1\}$) & $-y\log\hat{y}-(1-y)\log(1-\hat{y})$ \\[2pt]
      \textbf{Softmax + CE} & Class. Multiclasse & $-\sum_{i} y_i \log \hat{y}_i$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.1cm}

  \begin{block}{Esempio: Object Detector (es. YOLO)}
    \scriptsize
    Combinazione di più loss in un unico modello:
    \begin{itemize}
      \item \textbf{MSE} per la posizione della bounding box.
      \item \textbf{Cross-Entropy Binaria} per la presenza dell'oggetto.
      \item \textbf{Softmax} per la classe dell'oggetto.
    \end{itemize}
  \end{block}
\end{frame}

\section{Addestrare una rete profonda}
% ==================================================================
\begin{frame}{Gradient Descent}
  \small
  \begin{block}{Regola di aggiornamento}
    Ogni peso si sposta nella direzione \textbf{opposta al gradiente}:
    \[ w_{ij} := w_{ij} - \eta\,\frac{\partial E}{\partial w_{ij}} \]
  \end{block}

  \vspace{-0.1cm}
  
  \begin{itemize}
    \item \textbf{Derivata parziale} $\frac{\partial E}{\partial w_{ij}}$: indica quanto varia l'errore $E$ modificando il solo peso $w_{ij}$ (con gli altri costanti).
    \item \textbf{Learning rate ($\eta$)}: regola l'ampiezza del passo. Se troppo grande sbalza oltre il minimo, se troppo piccolo la convergenza è lentissima.
    \item \textbf{Backpropagation}: calcola le derivate parziali per tutti i pesi.
    \item \textbf{Ottimizzatori avanzati}: Mini-batch, Momentum, RMSprop, Adam.
  \end{itemize}
\end{frame}



\begin{frame}{Backpropagation}
  Si propaga l'errore \textbf{all'indietro}, dal layer di output verso l'input,
  con la \emph{regola della catena}. Per ogni unità si calcola un errore
  $\delta_i^{(l)}$:
  \begin{align*}
    \delta_i^{(L)} &= \ell'\big(h_\Theta(\mathbf{x}^{(i)})\big)
      && \text{errore sull'output} \\
    \delta_i^{(l)} &= \Big(\sum_k \delta_k^{(l+1)}\,\theta_{ki}^{(l)}\Big)\,
      \varphi'\big(a_i^{(l)}\big)
      && \text{errore riportato indietro} \\
    \frac{\partial J(\Theta)}{\partial \theta_{ij}^{(l)}} &=
      a_j^{(l)}\,\delta_i^{(l+1)}
      && \text{gradiente di ogni peso}
  \end{align*}
  \begin{itemize}
    \item L'errore di un'unità nascosta usa la \textbf{derivata della sua
          funzione di attivazione}.
    \item Un passo di training = forward propagation + backpropagation +
          aggiornamento dei pesi.
  \end{itemize}
\end{frame}

\begin{frame}{Gradienti instabili: vanishing ed exploding}
  In una catena di unità con bias $b_i$, pesi $w_i$, attivazione $\sigma$ e
  costo $C$, il gradiente di un layer iniziale è un \textbf{prodotto}:
  \[ \frac{\partial C}{\partial b_1} = \sigma'(z_1)\, w_2\, \sigma'(z_2)\,
     w_3\, \sigma'(z_3)\, w_4\, \sigma'(z_4)\, \frac{\partial C}{\partial a_4} \]
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Vanishing}
        Se ogni prodotto $w_i\,\sigma'(z_i) < 1$ il gradiente \textbf{svanisce}.
        Con la logistica $0 \le \sigma'(z) \le 0.25$ e pesi da
        $\mathcal{N}(0,1)$ succede facilmente.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Exploding}
        Se ogni prodotto $w_i\,\sigma'(z_i) > 1$, per esempio con pesi
        inizializzati grandi, il gradiente \textbf{esplode} verso i primi layer.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  In entrambi i casi i layer imparano a \textbf{velocità diverse}: i primi
  layer, con i prodotti più lunghi, sono i più colpiti.
\end{frame}

\begin{frame}{Inizializzazione dei pesi}
  \small
  Con input standardizzati e pesi da $\mathcal{N}(0,1)$, la somma
  $z = \theta_1x_1 + \dots + \theta_nx_n$ ha media 0 e \textbf{varianza $n$}
  (deviazione standard $\sqrt{n}$). Per $n$ grande $z$ tende a esplodere o
  svanire.
  \begin{block}{Soluzione: riscalare i pesi di ogni layer}
    Si mantiene deviazione standard 1 in ogni layer moltiplicando per un
    fattore che dipende da $n^{(l-1)}$, il numero di input del layer:
    \begin{align*}
      \theta_{ij}^{(l)} &= \mathcal{N}(0,1)\cdot\sqrt{1/n^{(l-1)}}
        && \text{attivazione lineare} \\
      \theta_{ij}^{(l)} &= \mathcal{N}(0,1)\cdot\sqrt{2/n^{(l-1)}}
        && \text{ReLU (inizializzazione di \textbf{He})} \\
      \theta_{ij}^{(l)} &= \mathcal{N}(0,1)\cdot\sqrt{2/\big(n^{(l-1)}+n^{(l)}\big)}
        && \tanh\ \text{(inizializzazione di \textbf{Xavier})}
    \end{align*}
  \end{block}
  Sigmoidi con inizializzazione sbagliata $\Rightarrow$ gradienti instabili;
  \textbf{ReLU + He} è la combinazione standard.
\end{frame}

\begin{frame}{Overfitting e regolarizzazione L2}
  La regolarizzazione riduce l'\textbf{overfitting}, problema centrale nel deep
  learning, dove le reti hanno moltissimi parametri.
  \begin{block}{Costo regolarizzato}
    \[ J(\Theta) = \frac{1}{m}\sum_{i=1}^{m} \ell\big(\hat{y}^{(i)},y^{(i)}\big)
       + \frac{\lambda}{2m}\sum_{l=1}^{L}\big\lVert\Theta^{(l)}\big\rVert^2 \]
  \end{block}
  \begin{itemize}
    \item $\lambda \uparrow \;\Rightarrow\; \theta_{ij}^{(l)} \downarrow$: molti
          pesi vicini a 0, reti più semplici e sparse, meno overfitting.
    \item \textbf{Effetto sull'attivazione}: con pesi piccoli l'argomento $z$ è
          vicino a 0, dove la funzione di attivazione è quasi
          \textbf{lineare}. La rete approssima solo funzioni meno complesse.
  \end{itemize}
  \begin{center}
    \small Nel progetto la regolarizzazione L2 è il \emph{weight decay} passato
    all'ottimizzatore.
  \end{center}
\end{frame}

\begin{frame}{Dropout, data augmentation, early stopping}
  \small
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \begin{block}{Dropout}
        Alternativa alla L2: ogni nodo viene \textbf{eliminato} con
        probabilità $p$.
        \begin{itemize}
          \item $p$ unica, diversa per layer (più alta nei layer grandi) o solo
                su alcuni layer.
          \item Reti più semplici; nessuna unità può contare su una singola
                feature, quindi i pesi si distribuiscono.
          \item \textbf{Mai a test time.}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.28\textwidth}
      \begin{block}{Data augmentation}
        Nuovi esempi trasformando quelli esistenti: flip, crop, zoom,
        rotazioni, distorsioni, variazioni di colore.
        \\[4pt]
        Riduce l'overfitting ed è naturale nella visione.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Early stopping}
        Si ferma il training quando l'errore sul validation ricomincia a
        salire.
        \\[4pt]
        Limite: ottimizza il costo e riduce la varianza \emph{insieme}; la L2
        tiene separati i due compiti (\emph{ortogonalità}).
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.05cm}
  \begin{center}
    \footnotesize Nel progetto: augmentation nel detector, dropout dentro
    BERT, scelta del checkpoint sul validation in entrambi.
  \end{center}
\end{frame}

\begin{frame}{Mini-batch gradient descent}
  Le reti profonde si addestrano su \textbf{big data}: calcolare il gradiente
  su tutti gli $m$ esempi prima di ogni aggiornamento può essere proibitivo.
  Si divide il training set in $k$ mini-batch e si aggiornano i pesi dopo
  \textbf{ogni} mini-batch; un'\textbf{epoca} è un passaggio completo.
  \vspace{0.2cm}
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    Dimensione & Metodo & Comportamento \\
    \midrule
    $m$ & batch GD & può essere lento \\
    $1$ & stochastic GD (SGD) & veloce, ma perde la vettorizzazione \\
    intermedia & \textbf{mini-batch GD} & il compromesso usato in pratica \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.2cm}
  \begin{itemize}
    \item Batch piccoli: convergenza rapida ma soluzione sub-ottima; batch
          grandi: convergenza lenta ma soluzione migliore.
    \item Regole pratiche: con $m \le 2000$ batch GD; altrimenti batch da 64,
          128, 256 o 512, che \textbf{stiano in memoria} della CPU/GPU.
  \end{itemize}
\end{frame}

\begin{frame}{Medie mobili esponenziali}
  Sono il componente chiave degli ottimizzatori per reti profonde:
  \[ v_t = \beta\, v_{t-1} + (1-\beta)\,\theta_t
     = (1-\beta)\theta_t + (1-\beta)\beta\,\theta_{t-1}
     + (1-\beta)\beta^2\theta_{t-2} + \dots \]
  \begin{itemize}
    \item I termini più vecchi pesano meno, perché $\beta^t$ decresce con $t$.
    \item $\frac{1}{1-\beta}$ approssima l'ampiezza della finestra: $\beta=0.5$
          circa 2 valori, $\beta=0.9$ circa 10, $\beta=0.98$ circa 50.
  \end{itemize}
  \begin{block}{Bias correction}
    Partendo da $v_0=0$, con $\beta=0.98$ la curva parte troppo in basso. Si
    corregge ogni valore:
    \[ v'_t = \frac{v_t}{1-\beta^{t}} \]
    All'inizio il fattore normalizza la somma pesata; per $t$ grande non ha
    più effetto.
  \end{block}
\end{frame}

\begin{frame}{Momentum, RMSprop, Adam}
  \footnotesize
  Sia $dw_{ij} = \partial E/\partial w_{ij}$, calcolato sul mini-batch.
  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \begin{block}{Momentum}
        Media mobile del \textbf{gradiente}:
        \[ \nu_{ij} = \beta\,\nu_{ij} + (1-\beta)\,dw_{ij} \]
        \[ w_{ij} = w_{ij} - \eta\,\nu_{ij} \]
        Aiuta a \textbf{uscire dai minimi locali}. Con $\beta$ grande serve
        $\eta$ piccolo; di solito senza bias correction.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{RMSprop}
        Media mobile del \textbf{quadrato} del gradiente:
        \[ \nu_{ij} = \beta\,\nu_{ij} + (1-\beta)\,dw_{ij}^2 \]
        \[ w_{ij} = w_{ij} - \eta\,\frac{dw_{ij}}{\sqrt{\nu_{ij}}+\epsilon} \]
        Normalizza le fluttuazioni: i valori grandi diventano più piccoli e
        viceversa.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Adam (\emph{ADAptive Momentum})}
        Momentum + RMSprop con bias correction:
        \[ \nu_{ij} = \tfrac{1}{1-\beta_1^t}\big(\beta_1\nu_{ij} + (1-\beta_1)dw_{ij}\big) \]
        \[ \sigma_{ij} = \tfrac{1}{1-\beta_2^t}\big(\beta_2\sigma_{ij} + (1-\beta_2)dw_{ij}^2\big) \]
        \[ w_{ij} = w_{ij} - \eta\,\frac{\nu_{ij}}{\sqrt{\sigma_{ij}}+\epsilon} \]
        $\beta_1=0.9$, $\beta_2=0.999$, $\epsilon=10^{-8}$; si regola $\eta$.
      \end{block}
    \end{column}
  \end{columns}
  \begin{center}
    Adam è l'ottimizzatore di \textbf{entrambi} gli esercizi (nell'Es.\ 2 nella
    variante AdamW).
  \end{center}
\end{frame}

\begin{frame}{Batch Normalization}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Idea}
        Come per gli input, si normalizzano i \textbf{nodi interni} $z_i$
        (prima dell'attivazione), con media $\mu_i$ e varianza $\sigma_i^2$
        calcolate su ogni \textbf{mini-batch}:
        \[ \hat{z}_i^{(k)} = \frac{z_i^{(k)}-\mu_i}{\sqrt{\sigma_i^2+\epsilon}},
           \qquad
           \tilde{z}_i^{(k)} = \gamma_i\,\hat{z}_i^{(k)} + \beta_i \]
        $\gamma_i,\beta_i$ si apprendono: normalizzare a media 0 e varianza 1
        non conviene, perché lì l'attivazione è quasi lineare.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Covariate shift}: l'input di un layer profondo dipende da
              tutti i pesi precedenti e cambia molto durante il training. BN ne
              mantiene fisse media e varianza.
        \item Media e varianza del mini-batch aggiungono un piccolo rumore, che
              fa da \textbf{regolarizzazione} (più forte con batch piccoli).
        \item La normalizzazione elimina il bias: con BN non servono bias
              unit.
        \item \textbf{A test time} si usano le medie mobili esponenziali di
              $\mu_i$ e $\sigma_i^2$ calcolate durante il training.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \begin{center}
    Nel detector ogni blocco è \texttt{Conv (senza bias) $\to$ BN $\to$ ReLU}.
  \end{center}
\end{frame}

\begin{frame}{Dalla teoria ai due esercizi}
  \begin{block}{Quello che serve a entrambi}
    Rete profonda con attivazioni \textbf{ReLU}, loss di \textbf{cross-entropy},
    backpropagation, mini-batch con \textbf{Adam}, normalizzazione,
    regolarizzazione e scelta del modello sul \textbf{validation}.
  \end{block}
  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Object detection}
        Immagini $\to$ rete \textbf{convoluzionale}, che sfrutta la struttura
        spaziale dei pixel. \emph{Architettura YOLO.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Sentiment analysis}
        Frasi $\to$ \textbf{Transformer}, che mette in relazione ogni parola con
        tutte le altre. \emph{Modello BERT.}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{center}
    \small In entrambi si parte da una rete \textbf{pre-addestrata}
    (\emph{transfer learning}).
  \end{center}
\end{frame}

\section{Object Detection (YOLO)}
% ==================================================================

\begin{frame}{Reti convoluzionali: la convoluzione}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Una rete densa su immagini grandi è \textbf{ingestibile} (troppi pesi).
      \vspace{0.2cm}

      La \textbf{convoluzione} fa scorrere un piccolo \textbf{filtro} (kernel)
      sull'immagine, calcolando un prodotto scalare a ogni posizione $\to$
      produce una \emph{feature map}.
      \vspace{0.2cm}
      \begin{block}{Filtri appresi, non progettati}
        Un filtro può rilevare i \textbf{bordi} (verticali, orizzontali).
        Invece di costruirlo a mano (Sobel, Scharr), si \alert{apprendono i pesi
        del filtro} con la backpropagation.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\textwidth]{cnn_edge.pdf}\\[4pt]
      {\footnotesize Un filtro verticale evidenzia i bordi verticali.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{CNN: padding, stride, pooling}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Iperparametri della convoluzione}
        \begin{itemize}
          \item \textbf{Padding}: bordo di zeri; \emph{same} mantiene la
                dimensione, \emph{valid} la riduce.
          \item \textbf{Stride}: di quanti passi scorre il filtro.
          \item \textbf{Più filtri} $\to$ più feature map (profondità).
                Su immagini RGB: convoluzione su ogni canale, poi somma.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Pooling}
        Il \textbf{max pooling} tiene il valore massimo di ogni regione:
        riduce la dimensione e agisce da rilevatore di feature.
      \end{block}
      \vspace{0.1cm}
      \textbf{Architettura tipica}:
      \[ \text{CONV} \to \text{POOL} \to \dots \to \text{FC} \]
      Andando in profondità: più filtri, dimensioni spaziali più piccole.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Perché la convoluzione funziona}
  \begin{block}{Due proprietà chiave}
    \begin{itemize}
      \item \textbf{Parameter sharing}: un rilevatore di feature (es.\ un bordo)
            è utile in \textbf{tutta} l'immagine $\to$ pochissimi parametri, che
            \emph{non dipendono} dalla dimensione dell'immagine.
      \item \textbf{Sparsità delle connessioni}: ogni output dipende solo da una
            \textbf{piccola regione} dell'input $\to$ meno parametri, meno
            overfitting.
    \end{itemize}
  \end{block}
  \begin{center}
    \small Nel detector: la ResNet riduce $416\times416$ fino alla griglia
    $26\times26$ (stride 16); il ``collo'' aggiunge due blocchi
    \texttt{Conv $\to$ BN $\to$ ReLU} a risoluzione costante.
  \end{center}
\end{frame}

\begin{frame}{Il problema: localizzazione con bounding box}
  \small
  \begin{block}{Object detection}
    Data un'immagine, individuare \textbf{quali} oggetti contiene e \textbf{dove}
    si trovano, disegnando una \emph{bounding box} attorno a ciascuno.
  \end{block}
  \vspace{0.15cm}
  \textbf{Dataset: Object365} (365 classi, 600k immagini, 10M box).
  \begin{itemize}
    \item Volume incompatibile con un PC $\Rightarrow$ \alert{subsampling} su due assi.
    \item \textbf{Immagini}: 15 shard scaricati $\to$ \textbf{14.361} immagini (119.215 box),
          divise in train 11.467 / validation 1.395 / test 1.499.
    \item \textbf{Classi}: da 365 a \textbf{8}, scelte \emph{misurando} frequenza e dimensione:
          Person, Chair, Table, Cabinet, Car, Lamp, Picture, Monitor.
    \item Categorie ``sorelle'' \textbf{unite} (Car + SUV + Van + Pickup): altrimenti un SUV
          sarebbe sfondo accanto a un'auto identica etichettata come oggetto.
  \end{itemize}
  \begin{center}
    \small Tecnica scelta: \textbf{YOLO}. Griglia, anchor, loss, IoU e NMS sono
    \emph{implementati a mano} (nessuna libreria di detection).
  \end{center}
\end{frame}

\begin{frame}{Dalla sliding window a YOLO}
  \small
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Sliding window}: si fa scorrere una finestra e si classifica ogni ritaglio.
      \begin{itemize}
        \item Problema principale: \alert{costo computazionale}.
        \item Soluzione: implementarla \textbf{convoluzionalmente}.
      \end{itemize}
      \vspace{0.1cm}
      \textbf{YOLO}: griglia $S\times S$; ogni oggetto assegnato a \textbf{una cella}
      (quella del suo centro), che ne predice box e classe.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Anchor box}
        Per ogni cella, $n$ forme predefinite. Se più oggetti cadono nella stessa
        cella, ognuno va all'anchor più simile per forma (IoU).
      \end{block}
      \centering
      \includegraphics[width=0.72\textwidth]{anchor.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Distribuzione del dataset}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \vspace{0.4cm}
      Sottoinsieme \textbf{sbilanciato}: \textbf{Person} è il 51\% delle box,
      \textbf{Monitor} meno del 2\% $\Rightarrow$ prestazioni attese
      \alert{inferiori sulle classi rare}.
      \vspace{0.3cm}

      \begin{block}{Regioni ignorate (\emph{don't care})}
        Box \textbf{crowd} (una box su un gruppo di oggetti, il 15\% delle
        annotazioni) e oggetti \textbf{più piccoli di una cella}: etichetta
        ``?'', niente obiettivi né penalità. Cancellarli insegnerebbe che lì \emph{non c'è nulla}.
      \end{block}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\textwidth]{classi_new.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Architettura della rete}
  \small
  Backbone convoluzionale con downsampling $\times 32$, seguito da una
  \textbf{testa di detection} $1\times1$ scritta da noi.
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      {\footnotesize
      \begin{semiverbatim}
Input  3 x 416 x 416
  ResNet34 troncata a layer3 (stride 16)
             -> 256 x 26 x 26   (griglia S)
  Collo: 2 x (conv 3x3 + BN + ReLU)
  Testa 1x1 -> A*(5+C) = 65 canali
Output (5, 26, 26, 13)
      \end{semiverbatim}}
      \vspace{0.05cm}
      \begin{itemize}
        \item Rete \textbf{completamente convoluzionale}.
        \item \textbf{3.380} box candidate per immagine ($5\times26\times26$).
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Backbone: transfer learning}
        \alert{\textbf{ResNet34 pre-addestrata}} su ImageNet e fine-tunata.
        Alternative provate: ResNet18 (--0,05 di mAP) e un backbone scritto
        interamente da zero.
      \end{block}
      {\footnotesize Anchor, target, loss, IoU e NMS restano \textbf{implementati a
      mano}: cambia solo l'estrattore di feature.}
      \vspace{0.08cm}
      {\footnotesize Il \textbf{troncamento} decide il compromesso: stride piccolo
      $\to$ griglia fine ma campo recettivo ridotto; stride grande $\to$ vede più
      contesto ma celle grossolane.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{La loss YOLO}
  \small
  La rete predice \emph{logit grezzi}, decodificati come (YOLOv2):
  \[
    b_x=\sigma(t_x)+c_x,\quad b_y=\sigma(t_y)+c_y,\quad
    b_w=a_w e^{t_w},\quad b_h=a_h e^{t_h}
  \]
  \begin{block}{Quattro componenti}
    \centering
    \begin{tabular}{llll}
      \toprule
      Componente & Funzione & Peso & Si applica a \\
      \midrule
      Coordinate & MSE & $\lambda=5$ & anchor responsabili \\
      Objectness & BCE & $1$ & anchor responsabili \\
      No-object & BCE & $\lambda=10$ & anchor vuoti \emph{e non ignorati} \\
      Classe & Cross-entropy & $1$ & anchor responsabili \\
      \bottomrule
    \end{tabular}
  \end{block}
  Su 3.380 box quasi tutte sono vuote: il termine no-object è normalizzato come
  \textbf{media sulle celle vuote} (non dipende dalla griglia) e pesato $\lambda=10$,
  il rapporto del YOLO originale. Troppo basso $\to$ oggetti ovunque; troppo alto
  $\to$ \alert{``non c'è mai niente''}.
\end{frame}

\begin{frame}{Inferenza: non-max suppression}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Ogni box: \textbf{score} $=$ objectness $\times$ prob.\ classe.
      \begin{enumerate}
        \item Scarta le box sotto soglia (0,25).
        \item \textbf{NMS} per classe.
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Non-max suppression}
        {\footnotesize
        ordina le box per score\\
        finché restano box:\\
        \quad prendi quella con score max $\to$ predizione\\
        \quad scarta le altre con $\text{IoU}>$ soglia}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  NMS \emph{per classe}: due oggetti di classi diverse possono sovrapporsi (una
  persona con un cappello); due box sovrapposte della stessa classe no.
\end{frame}

\begin{frame}{Come si misura un detector}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Una predizione è \textbf{corretta} se ha IoU $\geq 0{,}5$ con un oggetto della
      stessa classe non ancora abbinato; le altre sono falsi positivi.
      \[ P=\frac{TP}{TP+FP}\qquad R=\frac{TP}{N_{ogg}}\qquad F_1=\frac{2PR}{P+R} \]
      \begin{alertblock}{La trappola della soglia fissa}
        Lo stesso modello: \textbf{F1 0,145} a soglia 0,25 e \textbf{0,249} a 0,65.
        Confrontare modelli a soglia fissa è ingannevole.
      \end{alertblock}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{mAP@0.5}
        Si fa variare la soglia, si costruisce la curva \textbf{precision--recall} e
        se ne prende l'\textbf{area} (AP), mediata sulle classi.
        \textbf{Non dipende dalla soglia} $\Rightarrow$ è la metrica con cui si
        confrontano le varianti e si sceglie il checkpoint.
      \end{block}
      {\footnotesize La soglia d'uso (0,9) è poi quella che massimizza l'F1 sul
      validation, salvata nel checkpoint.}
    \end{column}
  \end{columns}
\end{frame}

\section{Sentiment Analysis (Transformer)}
% ==================================================================

\begin{frame}{Il problema: classificare frasi}
  \begin{block}{Sentiment analysis}
    Classificare una sequenza testuale come \textbf{positiva} o \textbf{negativa},
    usando un'architettura \textbf{transformer}.
  \end{block}
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Dataset}: SST-2 (frasi brevi etichettate pos/neg, da HuggingFace).
    \item Corrisponde alla traccia (``frasi positive o negative''); nessun token Kaggle richiesto.
    \item Tre split: \textbf{train}; \textbf{validation} (2000 frasi ricavate dal
          train) per scegliere il checkpoint; \textbf{test} (le 872 frasi del
          validation ufficiale) usato \emph{una sola volta} per i risultati.
  \end{itemize}
  \begin{center}
    \small Il codice supporta comunque un CSV Kaggle via \texttt{-{}-dataset csv}.
  \end{center}
\end{frame}

\begin{frame}{Dai limiti delle RNN all'attention}
  \textbf{Le RNN} codificano bene la storia recente, ma faticano su contesti
  lontani e sono \textbf{sequenziali} (lente).
  \begin{block}{Esempio delle slide}
    \emph{``I grew up in France, \dots\ thus I speak ??''} --- la parola da predire
    (\emph{French}) dipende da un termine molto \textbf{lontano} nella sequenza.
  \end{block}
  \vspace{0.2cm}
  La risposta è il \textbf{meccanismo di attention}: per produrre la
  rappresentazione di una parola, la si mette in relazione con \textbf{tutte} le
  altre e si pesa quali sono rilevanti, indipendentemente dalla distanza.
\end{frame}

\begin{frame}{Self-attention}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      L'output per ogni parola è una \textbf{somma pesata} delle rappresentazioni
      di tutte le parole; i pesi misurano quanto ognuna è \emph{rilevante}.
      \begin{block}{Proprietà (dalle slide)}
        \begin{itemize}
          \item Calcolo \textbf{parallelo} di tutti gli output (non sequenziale).
          \item \emph{Permutation equivariant}: è più un modello di
                \textbf{insiemi} che di sequenze $\Rightarrow$ serve il
                \textbf{positional encoding} per reintrodurre l'ordine.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=0.92\textwidth]{attention.pdf}\\[3pt]
      {\footnotesize \emph{cattivo} mette attenzione su \emph{non}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Query, Key, Value e multi-head}
  \begin{block}{Tre ruoli, tre proiezioni lineari apprese}
    \begin{itemize}
      \item \textbf{Query}: il vettore che viene confrontato con tutti gli altri.
      \item \textbf{Key}: il vettore contro cui la query è confrontata.
      \item \textbf{Value}: ciò che si somma (pesato) per formare l'output.
    \end{itemize}
  \end{block}
  \[ \mathrm{Attention}(Q,K,V)=\mathrm{softmax}\!\left(\frac{QK^{\top}}{\sqrt{d}}\right)V \]
  \begin{itemize}
    \item Divisione per $\sqrt{d}$: \textbf{scaled} attention (stabilità).
    \item \textbf{Multi-head}: più ``teste'' in parallelo, ciascuna cattura un
          tipo diverso di relazione tra le parole.
  \end{itemize}
\end{frame}

\begin{frame}{Il blocco Transformer}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Un blocco impila due sotto-strati}
        \begin{enumerate}
          \item \textbf{Multi-head self-attention}
          \item \textbf{Rete feed-forward} (applicata a ogni posizione)
        \end{enumerate}
        Ciascuno preceduto da una \textbf{layer normalization} e seguito da una
        \textbf{connessione residua} (skip, \emph{Pre-LN}):
        \[ x + \mathrm{Sublayer}\big(\mathrm{LayerNorm}(x)\big) \]
      \end{block}
      Impilando più blocchi si ottiene il Transformer.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Positional encoding}
        La self-attention è \emph{permutation equivariant}: si \textbf{somma} a
        ogni token un vettore che ne codifica la \textbf{posizione}, così il
        modello conosce l'ordine.
      \end{block}
      \begin{block}{Layer normalization}
        Come la batch norm, ma normalizza sulle \textbf{feature di ogni token}
        (non sul batch).
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Il Transformer e BERT}
  \begin{block}{Transformer}
    Un modello che usa principalmente la \textbf{self-attention} per propagare
    l'informazione tra le unità (token di testo, ma anche pixel: Vision
    Transformer).
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{BERT}: transformer \textbf{encoder-only} \emph{bidirezionale},
      pre-addestrato con il \textbf{masking} (predire token nascosti).
      Adatto a \emph{comprendere} una frase, non a generarla.
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Il nostro uso}
        \textbf{Fine-tuning} (transfer learning): si parte da BERT
        pre-addestrato e si aggiunge una testa di classificazione binaria.
        Modello: \textbf{DistilBERT}, un BERT più leggero (6 strati invece di 12).
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Analisi qualitativa: il caso della negazione}
  Frasi costruite ad hoc:
  \begin{block}{Output del modello}
    {\footnotesize\ttfamily
    {}[positive~~99.9\%] I absolutely loved this film\\
    {}[negative~~99.6\%] the worst purchase I have ever made\\
    {}[positive~~99.8\%] It is not bad at all, quite enjoyable\\
    {}[negative~~83.2\%] the acting was fine, the story less so}
  \end{block}
  \begin{itemize}
    \item \emph{``not bad''}: contiene \emph{bad} ma è classificata \textbf{positiva} al 99,8\%.
          L'attention collega \emph{bad} a \emph{not} --- è l'\alert{esempio della slide 11}.
    \item Frase a sentimento misto $\Rightarrow$ confidenza più bassa (83\%): il modello \emph{segnala} l'incertezza.
  \end{itemize}
\end{frame}

\begin{frame}{Aderenza al programma del corso}
  \footnotesize
  \textbf{Nucleo del progetto: tecniche viste a lezione.} YOLO con griglia, anchor
  scelti a mano e assegnati per IoU, NMS; etichette \emph{don't care} (le nostre
  \textbf{regioni ignorate}); CNN, ResNet34, batch norm; \textbf{transfer learning}
  (backbone pre-addestrato, fine-tuning di BERT); transformer e self-attention;
  Adam, regolarizzazione L2, early stopping; data augmentation.
  \begin{block}{Aggiunto oltre il programma, con motivazione}
    \begin{itemize}
      \setlength{\itemsep}{1pt}
      \item \textbf{mAP@0.5} (Es.\ 1): precision e recall dipendono dalla soglia,
            quindi non confrontano modelli calibrati diversamente.
      \item \textbf{Learning rate variabile}: decadimento coseno (Es.\ 1, +0,008 di
            mAP, misurato); warmup e decadimento lineare (Es.\ 2, standard per il
            fine-tuning di BERT: evita passi grandi sui pesi pre-addestrati).
      \item \textbf{AdamW} (Es.\ 2): Adam con il weight decay applicato separatamente
            dal gradiente, ottimizzatore predefinito del Trainer HuggingFace.
      \item \textbf{DistilBERT} e \textbf{fp16} (Es.\ 2): un BERT più piccolo
            ottenuto per distillazione e calcoli a mezza precisione, solo per
            ridurre tempo e memoria.
    \end{itemize}
